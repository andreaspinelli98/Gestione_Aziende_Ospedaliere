CREATE DOMAIN COSTO AS VARCHAR CHECK
(value = 'ssn' OR value = 'privato');

CREATE DOMAIN INTEST AS VARCHAR CHECK
(value = 'interno' OR value = 'esterno');

CREATE DOMAIN FASCIA AS VARCHAR CHECK
(value = 'verde' OR value = 'gialla' OR value = 'rossa');

CREATE DOMAIN ORA AS VARCHAR CHECK
(value = '6-14' OR value = '14-22' OR value = '22-6');

--controllo  che domini seguenti abbiano gli apici
CREATE DOMAIN GRADO AS VARCHAR CHECK
(value='A' OR value='B' OR value='C');

CREATE DOMAIN AREA AS VARCHAR CHECK
(value='sanità pubblica' OR value='pediatria' OR value='psichiatria' OR value='area critica' OR value='geriatria');

CREATE TABLE ospedale (
	codosp VARCHAR (20) PRIMARY KEY,
    	nomeosp VARCHAR (20) NOT NULL,
    	indirizzosp VARCHAR(20) UNIQUE,
	tel VARCHAR (10) UNIQUE
);

CREATE TABLE reparto (
	codiceosp VARCHAR (20),
	nomerep VARCHAR (20) NOT NULL,
    	telrep VARCHAR (10) UNIQUE,
    	orariovisite VARCHAR (20) NOT NULL,
    	PRIMARY KEY (codiceosp, nomerep),
	FOREIGN KEY (codiceosp) REFERENCES ospedale(codosp) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE ambulatorio (
	nomeamb VARCHAR(20) NOT NULL,
	tipoamb INTEST,
	pianoamb VARCHAR(5),
	indirizzoamb VARCHAR(20) UNIQUE,
	telamb VARCHAR(10) UNIQUE,
	oraperturamb TIME NOT NULL,
	PRIMARY KEY(nomeamb)
);

CREATE TABLE stanzanalisi (
	nomeamb VARCHAR(20),
	numstanza INTEGER NOT NULL,
	PRIMARY KEY(nomeamb, numstanza),
	FOREIGN KEY (nomeamb) REFERENCES ambulatorio(nomeamb) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE stanzaricovero ( 
	codiceosp VARCHAR(20), 
	nomerep VARCHAR(20), 
	numstanza INTEGER NOT NULL, 
	piano VARCHAR(5) NOT NULL, 
	PRIMARY KEY (codiceosp, nomerep, numstanza), 
	FOREIGN KEY (codiceosp) REFERENCES ospedale(codosp) ON DELETE CASCADE ON UPDATE CASCADE, 
	FOREIGN KEY (codiceosp, nomerep) REFERENCES reparto(codiceosp, nomerep) ON DELETE CASCADE ON UPDATE CASCADE );


CREATE TABLE salaoperatoria (
	codiceosp VARCHAR(20),
   	nomereparto VARCHAR(20),
   	numsala INTEGER NOT NULL,
    	PRIMARY KEY (codiceosp, nomereparto, numsala),
    	FOREIGN KEY (codiceosp) REFERENCES ospedale(codosp) ON DELETE CASCADE ON UPDATE CASCADE,
    	FOREIGN KEY (codiceosp, nomereparto) REFERENCES reparto(codiceosp, nomerep) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE letto (
	codiceosp VARCHAR(20),
	nomereparto VARCHAR(20),
	numstanza INTEGER,
	numletto INTEGER NOT NULL,
	statoletto VARCHAR(10) NOT NULL,
	PRIMARY KEY (codiceosp, nomereparto, numstanza, numletto),
	FOREIGN KEY (codiceosp, nomereparto) REFERENCES reparto(codiceosp, nomerep) ON DELETE CASCADE ON UPDATE CASCADE,
	FOREIGN KEY (codiceosp, nomereparto, numstanza) REFERENCES stanzaricovero(codiceosp, nomerep, numstanza) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE paziente (
	tesserasanitaria VARCHAR(16) NOT NULL,
	nomepaz VARCHAR(20) NOT NULL,
	cognomepaz VARCHAR(20) NOT NULL,
	etàpaz INTEGER NOT NULL,
	datanascita DATE NOT NULL,
	indirizzopaz VARCHAR(20) NOT NULL,
	telpaz VARCHAR(10) UNIQUE,
	città VARCHAR(20) NOT NULL,
	datadimiss DATE,
	PRIMARY KEY (tesserasanitaria)
);

CREATE TABLE ricovero (
    codiceosp VARCHAR(20),
    nomereparto VARCHAR(20),
    numstanza INTEGER,
    numletto INTEGER not null,
    tessanit VARCHAR(20),
    id VARCHAR(20),
    datainizio DATE,
    datafine DATE,
    PRIMARY KEY (codiceosp, nomereparto, numstanza, numletto, tessanit, id),
    FOREIGN KEY (tessanit) REFERENCES paziente(tesserasanitaria) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (codiceosp, nomereparto) REFERENCES reparto(codiceosp, nomerep) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (codiceosp, nomereparto, numstanza) REFERENCES stanzaricovero(codiceosp, nomerep, numstanza) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (codiceosp, nomereparto, numstanza, numletto) REFERENCES letto(codiceosp, nomereparto, numstanza, numletto) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE patologia (
	nome VARCHAR(50) NOT NULL,
	PRIMARY KEY (nome)
);

CREATE TABLE diagnosi (
	tessanit VARCHAR(16),
	malattia VARCHAR(50),
	PRIMARY KEY (tessanit, malattia),
	FOREIGN KEY (tessanit) REFERENCES paziente(tesserasanitaria) ON DELETE CASCADE ON UPDATE CASCADE,
	FOREIGN KEY (malattia) REFERENCES patologia(nome) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE prontosoccorso ( 
	codiceosp VARCHAR(5), 
	nomeps VARCHAR(50), 
	PRIMARY KEY (codiceosp), 
	FOREIGN KEY (codiceosp) REFERENCES ospedale(codosp) ON DELETE CASCADE ON UPDATE CASCADE 
);

CREATE TABLE personaleammin (
	codiceosp VARCHAR(20) NOT NULL,
	rep VARCHAR(20) NOT NULL,
    	codfisc VARCHAR(16) UNIQUE NOT NULL,
	nome VARCHAR(20) NOT NULL,
	cognome VARCHAR(20) NOT NULL,
	tel VARCHAR(10) NOT NULL,
	livello GRADO,
	PRIMARY KEY (codiceosp, rep, codfisc),
	FOREIGN KEY (codiceosp, rep) REFERENCES reparto(codiceosp, nomerep) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE infermiere (
codiceosp VARCHAR(20) NOT NULL,
	rep VARCHAR(20) NOT NULL,
    	codfisc VARCHAR(16) UNIQUE NOT NULL,
    	nome VARCHAR(20) NOT NULL,
    	cognome VARCHAR(20) NOT NULL,
    	tel VARCHAR(10) NOT NULL,
    	tipo AREA,
    	PRIMARY KEY (codiceosp, rep, codfisc),
    	FOREIGN KEY (codiceosp, rep) REFERENCES reparto(codiceosp, nomerep) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE medico (
    	codiceosp VARCHAR(20) NOT NULL,
rep VARCHAR(20) NOT NULL,
    	codfisc VARCHAR(16) UNIQUE NOT NULL,
    	nome VARCHAR(20) NOT NULL,
    	cognome VARCHAR(20) NOT NULL,
    	tel VARCHAR(10) NOT NULL,
    	anzianità INTEGER,
    	PRIMARY KEY (codiceosp, rep, codfisc),
    	FOREIGN KEY (codiceosp, rep) REFERENCES reparto(codiceosp, nomerep) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE turno (
	codiceosp VARCHAR(5),
	orario ORA,
	giorno VARCHAR(10) NOT NULL,
	PRIMARY KEY (codiceosp, orario, giorno),
	FOREIGN KEY (codiceosp) REFERENCES prontosoccorso(codiceosp) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE turnazioneinf (
    codfisc VARCHAR(16),
    codiceosp VARCHAR(5),
    fasciainf ORA,
    giorno VARCHAR(10) NOT NULL,
    PRIMARY KEY(codfisc, codiceosp, fasciainf, giorno),
FOREIGN KEY (codfisc) REFERENCES infermiere(codfisc) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (codiceosp, fasciainf, giorno) REFERENCES turno(codiceosp, orario, giorno) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE turnazionemed (
    codfisc VARCHAR(16),
    codiceosp VARCHAR(5),
    fasciamed ORA,
    giorno VARCHAR(10) NOT NULL,
    PRIMARY KEY(codfisc, codiceosp, fasciamed, giorno),
FOREIGN KEY (codfisc) REFERENCES medico(codfisc) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (codiceosp, fasciamed, giorno) REFERENCES turno(codiceosp, orario, giorno) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE primario (
    codiceosp VARCHAR(20) NOT NULL,
    rep VARCHAR(20) NOT NULL,
	cf VARCHAR(16),
    PRIMARY KEY (codiceosp, rep, cf),
    FOREIGN KEY (codiceosp, rep, cf) REFERENCES medico(codiceosp, rep, codfisc) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE viceprimario (
    codiceosp VARCHAR(20) NOT NULL,
    rep VARCHAR(20) NOT NULL,
	cf VARCHAR(16),
	assunzruolo DATE NOT NULL,
    PRIMARY KEY (codiceosp, rep, cf),
    FOREIGN KEY (codiceosp, rep, cf) REFERENCES medico(codiceosp, rep, codfisc) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE specializzazione (
    codiceosp VARCHAR(20) NOT NULL,
    rep VARCHAR(20) NOT NULL,
    cf VARCHAR(16),
    tipo VARCHAR(50) NOT NULL,
    PRIMARY KEY (codiceosp, rep, cf, tipo),
    FOREIGN KEY (codiceosp, rep, cf) REFERENCES primario(codiceosp, rep, cf) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE sostituzione (
    codiceospvice VARCHAR(20) NOT NULL,
    repvice VARCHAR(20) NOT NULL,
    cfvice VARCHAR(16),
    codiceospmed VARCHAR(20),
    repmed VARCHAR(20),
    sostituto VARCHAR(16),
    datainizio DATE NOT NULL,
    datafine DATE,
    PRIMARY KEY (codiceospvice, repvice, cfvice, codiceospmed, repmed, sostituto, datainizio),
    FOREIGN KEY (codiceospvice, repvice, cfvice) REFERENCES viceprimario(codiceosp, rep, cf) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (codiceospmed, repmed, sostituto) REFERENCES primario(codiceosp, rep, cf) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE prenotazione (
	tessanit VARCHAR(16),
	nomeamb VARCHAR(20),
	id VARCHAR(6) UNIQUE,
	datapren DATE NOT NULL,
	dataesame DATE NOT NULL,
	oraesame TIME NOT NULL,
	urgenza FASCIA,
	regimecosto COSTO,
	PRIMARY KEY (tessanit, nomeamb, id),
	FOREIGN KEY (tessanit) REFERENCES paziente(tesserasanitaria) ON DELETE CASCADE ON UPDATE CASCADE,
	FOREIGN KEY (nomeamb) REFERENCES ambulatorio(nomeamb) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE esame (
	ts VARCHAR(16),
	pren VARCHAR(6),
	nomeamb VARCHAR(20) NOT NULL,
	codesame VARCHAR(10) NOT NULL,
	descriz VARCHAR(50),
	PRIMARY KEY (ts, pren, nomeamb, codesame),
	FOREIGN KEY (ts, pren, nomeamb) REFERENCES prenotazione(tessanit, id, nomeamb)
);

CREATE TABLE esamespecialistico (
    cfmed VARCHAR(16),
    ts VARCHAR(16),
    pren VARCHAR(6),
	nomeamb VARCHAR(20),
    codesame VARCHAR(10),
    avvertenze VARCHAR(50),
    PRIMARY KEY (cfmed, ts, nomeamb, pren, codesame),
    FOREIGN KEY (cfmed) REFERENCES medico(codfisc) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (ts, nomeamb, pren, codesame) REFERENCES esame(ts, nomeamb, pren, codesame) ON DELETE CASCADE ON UPDATE CASCADE
);
